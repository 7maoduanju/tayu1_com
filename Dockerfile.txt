FROM python:3.12-slim-bookworm


ENV PYTHONDONTWRITEBYTECODE=1
ENV PYTHONUNBUFFERED=1


WORKDIR /app


RUN useradd \
    --system \
    --uid 10001 \
    --create-home \
    serverguard


COPY main.py /app/main.py


RUN mkdir -p \
    /app/data \
    /app/logs \
    && \
    chown -R \
    serverguard:serverguard \
    /app


USER serverguard


CMD [
    "python3",
    "/app/main.py"
]
